\usepackage[T1]{fontenc}\usepackage{newpxtext,newpxmath}  % same fonts as the notes
\begin{document}
\begin{tikzpicture}[>=Stealth, lab/.style={font=\small, text=ink}]
  \def\B{5.6}\def\tA{20}\def\tB{42}\def\rl{5.4}
  \colorlet{cA}{accent}\colorlet{cB}{astroblue}
  \coordinate (A1) at (0,0);
  \coordinate (A2) at (\B,0);
  % normals
  \foreach \x in {0,\B}{\draw[muted, dashed, line width=0.5pt] (\x,0) -- (\x,5.0);}
  \node[lab, text=muted, above] at (0,5.0) {normal};
  \node[lab, text=muted, above] at (\B,5.0) {normal};
  % rays and wavefronts, one colour per source
  \foreach \t/\c/\n in {\tA/cA/1,\tB/cB/2}{
    \pgfmathsetmacro{\sx}{sin(\t)}\pgfmathsetmacro{\sy}{cos(\t)}
    \pgfmathsetmacro{\Ls}{\B*sin(\t)}
    \coordinate (W\n) at ({\Ls*\sx},{\Ls*\sy});
    \draw[->, \c, line width=0.9pt] ({\rl*\sx},{\rl*\sy}) -- (A1);
    \draw[->, \c, line width=0.9pt] ({\B+\rl*\sx},{\rl*\sy}) -- (A2);
    \draw[dashed, \c, line width=0.8pt] (A2) -- (W\n);
    \draw[\c, line width=0.6pt] ($(W\n)+({0.26*\sy},{-0.26*\sx})$) -- ++({-0.26*\sx},{-0.26*\sy}) -- ++({-0.26*\sy},{0.26*\sx});
    \draw[\c, line width=2.6pt] (W\n) -- (A1);
    \node[lab, text=\c, anchor=west] at ({\B+\rl*\sx+0.15},{\rl*\sy}) {source \n};
  }
  % angles at both antennas
  \foreach \x in {0,\B}{
    \draw[cA, line width=0.5pt] (\x,3.0) arc (90:{90-\tA+2}:3.0);
    \node[lab, text=cA] at ({\x+3.35*sin(\tA/2)},{3.35*cos(\tA/2)}) {$\theta_1$};
    \draw[cB, line width=0.5pt] (\x,4.2) arc (90:{90-\tB+2}:4.2);
    \node[lab, text=cB] at ({\x+4.55*sin((\tA+\tB)/2+4)},{4.55*cos((\tA+\tB)/2+4)}) {$\theta_2$};
  }
  % path labels, left of the normal at A1
  \node[lab, text=cA, anchor=east] (l1) at (-0.35,1.0) {$\Delta L_1=B\sin\theta_1$};
  \draw[cA, line width=0.4pt] (l1.east) -- ($(W1)!0.5!(A1)$);
  \node[lab, text=cB, anchor=west] (l2) at ($(W2)+(0.5,0)$) {$\Delta L_2=B\sin\theta_2$};
  \draw[cB, line width=0.4pt] ($(W2)+(0.08,0)$) -- ($(W2)+(0.45,0)$);
  % baseline and antennas
  \draw[ink, line width=1.4pt] (A1) -- (A2);
  \fill[astroblue] (A1) circle (2.6pt);
  \fill[astroblue] (A2) circle (2.6pt);
  \node[lab, below=3pt] at (A1) {$A_1$};
  \node[lab, below=3pt] at (A2) {$A_2$};
  \node[lab, below=4pt] at (\B/2,0) {baseline $B$};
\end{tikzpicture}
