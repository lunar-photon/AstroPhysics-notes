\definecolor{colone}{RGB}{180,60,60}
\definecolor{coltwo}{RGB}{60,110,180}

\begin{document}
% FRB dispersion sweep, t(nu) = t_inf + 4.15 ms x DM x (nu/GHz)^-2, plotted from the formula.
% Time measured from the arrival at the top of the band, 1.5 GHz.
% Solid: DM = 375 pc cm^-3 (Lorimer burst). Dashed: DM = 30 pc cm^-3 (pulsar 1 kpc away, as in the text).
\pgfmathsetmacro{\K}{4.15}      % ms GHz^2 per pc cm^-3
\pgfmathsetmacro{\DMfrb}{375}
\pgfmathsetmacro{\DMpsr}{30}
\pgfmathsetmacro{\numax}{1.5}
\begin{tikzpicture}[font=\small]
  \begin{axis}[width=10.5cm, height=6.2cm, scale only axis,
      xmin=-45, xmax=440, ymin=1.2, ymax=1.5, axis lines*=left,
      xlabel={arrival time after the $1.5\,\mathrm{GHz}$ signal (ms)},
      ylabel={frequency (GHz)},
      xtick={0,100,200,300,400}, ytick={1.2,1.3,1.4,1.5},
      yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=1},
      tick style={muted}, every axis label/.style={font=\small, text=muted}, clip=false]
    % the two sweeps, drawn parametrically in frequency
    \addplot[colone, line width=1.4pt, variable=\f, domain=1.2:1.5, samples=80]
      ({\K*\DMfrb*(1/\f^2-1/\numax^2)}, {\f});
    \addplot[coltwo, line width=1.1pt, dashed, variable=\f, domain=1.2:1.5, samples=40]
      ({\K*\DMpsr*(1/\f^2-1/\numax^2)}, {\f});
    \draw[ink, line width=1.1pt, dotted] (axis cs:0,1.2) -- (axis cs:0,1.5);
    \node[ink, rotate=90, anchor=south, yshift=3pt] at (axis cs:0,1.35) {no dispersion};
    \node[colone, anchor=west] at (axis cs:250,1.395) {$375\,\mathrm{pc\,cm^{-3}}$};
    \node[coltwo, anchor=west] at (axis cs:40,1.46) {$30\,\mathrm{pc\,cm^{-3}}$};
  \end{axis}
\end{tikzpicture}
\end{document}
