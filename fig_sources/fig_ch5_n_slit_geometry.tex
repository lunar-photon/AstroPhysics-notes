\documentclass[tikz,border=6pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{xcolor}
\definecolor{accent}{HTML}{A13D2C}
\definecolor{accent2}{HTML}{2B5F6B}
\definecolor{muted}{HTML}{5A564F}
\definecolor{panel}{HTML}{F8F6F0}
\definecolor{linecol}{HTML}{D8D2C4}
\definecolor{astroblue}{HTML}{1B4F72}
\definecolor{ink}{HTML}{1E1C1A}
\usetikzlibrary{calc,arrows.meta,angles,quotes}
\usepackage[T1]{fontenc}\usepackage{newpxtext,newpxmath}  % same fonts as the notes
\begin{document}
% A flat wave meets a grating of N slits a distance d apart (four drawn). Rays leave every slit at
% the angle theta; the tilted wavefront through the top slit shows the lags d sin(theta), 2d sin(theta),
% 3d sin(theta). A lens of focal length f brings the parallel rays to one point P(theta) of its focal plane.
\begin{tikzpicture}[font=\small, >={Stealth[length=2.2mm]}]
  \def\th{15}\def\xl{5.6}\def\f{3.4}
  \pgfmathsetmacro{\yp}{\f*tan(\th)}
  % incoming flat wave
  \foreach \x in {-2.4,-1.8,-1.2,-0.6} \draw[linecol, line width=0.8pt] (\x,-2.6) -- (\x,1.1);
  \foreach \y in {0.75,-0.25,-1.25,-2.25} \draw[astroblue, ->, line width=0.6pt] (-2.8,\y) -- (-0.15,\y);
  \node[muted, anchor=south] at (-1.5,1.15) {flat wave};
  % grating bars between the slits
  \foreach \yb in {1.55,0.55,-0.45,-1.45,-2.45} \fill[muted!70] (-0.07,\yb-0.6) rectangle (0.07,\yb);
  \node[muted, anchor=north] at (0,-3.1) {grating};
  % spacing d
  \draw[muted, <->] (-0.35,0.75) -- node[left, inner sep=2pt] {$d$} (-0.35,-0.25);
  % rays to the lens, then to P
  \foreach \y in {0.75,-0.25,-1.25,-2.25} {
    \pgfmathsetmacro{\yl}{\y+\xl*tan(\th)}
    \draw[accent, line width=0.8pt] (0,\y) -- (\xl,\yl) -- (\xl+\f,\yp);
    \draw[accent, ->, line width=0.8pt] (0,\y) -- ($(0,\y)!0.45!(\xl,\yl)$);
  }
  % tilted wavefront through the top slit and the lags
  \draw[accent2, dashed, line width=0.8pt] (0,0.75) -- ++(\th-90:3.4);
  \foreach \k/\y in {1/-0.25,2/-1.25,3/-2.25} {
    \pgfmathsetmacro{\L}{\k*sin(\th)}
    \draw[ink, line width=2.2pt] (0,\y) -- ++(\th:\L);
  }
  \node[ink, anchor=west, inner sep=2pt] at ($(0,-0.25)+(\th:{sin(\th)})+(0.3,-0.3)$) {$d\sin\theta$};
  \node[ink, anchor=west, inner sep=2pt] at ($(0,-1.25)+(\th:{2*sin(\th)})+(0.3,-0.3)$) {$2d\sin\theta$};
  \node[ink, anchor=west, inner sep=2pt] at ($(0,-2.25)+(\th:{3*sin(\th)})+(0.3,-0.3)$) {$3d\sin\theta$};
  % angle theta at the top slit
  \draw[muted, dashed] (0,0.75) -- (1.5,0.75);
  \draw[muted] (1.2,0.75) arc[start angle=0, end angle=\th, radius=1.2];
  \node[muted] at ($(0,0.75)+(\th/2:1.45)$) {$\theta$};
  % lens and optical axis
  \draw[muted, dash dot, line width=0.4pt] (\xl-0.6,0) -- (\xl+\f+0.4,0);
  \fill[astroblue!12] (\xl,0) ellipse (0.3 and 2.7);
  \draw[astroblue, line width=0.7pt] (\xl,0) ellipse (0.3 and 2.7);
  \node[muted, anchor=north] at (\xl,-2.75) {lens};
  % focal plane and P
  \draw[ink, line width=0.8pt] (\xl+\f,-2.6) -- (\xl+\f,2.0);
  \node[muted, anchor=north] at (\xl+\f,-2.65) {focal plane};
  \fill[accent] (\xl+\f,\yp) circle (0.07);
  \node[accent, anchor=west] at (\xl+\f+0.1,\yp) {$P(\theta)$};
  \draw[muted, <->] (\xl,-2.15) -- node[above, inner sep=2pt] {$f$} (\xl+\f,-2.15);
\end{tikzpicture}
\end{document}
