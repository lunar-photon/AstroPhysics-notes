% Two equal slit patterns (sin(pi x)/(pi x))^2, x = theta in units of lambda/a, a separation s apart,
% for s = 1 (Rayleigh), 0.83 (Sparrow) and 0.5. Dashed: the two patterns; solid: the sum.
\begin{tikzpicture}[font=\small, >={Stealth[length=1.8mm]},
    declare function={sincsq(\x)=(abs(\x)<1e-4 ? 1 : (sin(deg(pi*\x))/(pi*\x))^2);}]
  \pgfplotsset{every axis/.append style={width=3.9cm, height=3.6cm, scale only axis, axis lines*=left,
      xmin=-1.6, xmax=1.6, ymin=0, ymax=2.05, xtick={-1,0,1}, ytick={0,0.5,1,1.5},
      tick style={muted}, every axis label/.style={text=muted}, xlabel={$\theta$ in units of $\lambda/a$},
      clip=false, samples=161}}
  \begin{axis}[name=a,  ylabel={$I/I_0$}]
    \addplot[accent, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x+1/2)};
    \addplot[astroblue, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x-1/2)};
    \addplot[ink, line width=0.9pt, domain=-1.6:1.6] {sincsq(x+1/2)+sincsq(x-1/2)};
    \draw[ink, <->] (axis cs:-1/2,1.8) -- (axis cs:1/2,1.8);
    \node[ink, anchor=south, inner sep=2pt] at (axis cs:0,1.8) {Rayleigh, $\lambda/a$};
  \end{axis}
  \begin{axis}[name=b, at={(a.east)}, anchor=west, xshift=1.1cm, yticklabels={}]
    \addplot[accent, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x+0.83/2)};
    \addplot[astroblue, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x-0.83/2)};
    \addplot[ink, line width=0.9pt, domain=-1.6:1.6] {sincsq(x+0.83/2)+sincsq(x-0.83/2)};
    \draw[ink, <->] (axis cs:-0.83/2,1.8) -- (axis cs:0.83/2,1.8);
    \node[ink, anchor=south, inner sep=2pt] at (axis cs:0,1.8) {Sparrow, $0.83\lambda/a$};
  \end{axis}
  \begin{axis}[name=c, at={(b.east)}, anchor=west, xshift=1.1cm, yticklabels={}]
    \addplot[accent, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x+0.5/2)};
    \addplot[astroblue, dashed, line width=0.6pt, domain=-1.6:1.6] {sincsq(x-0.5/2)};
    \addplot[ink, line width=0.9pt, domain=-1.6:1.6] {sincsq(x+0.5/2)+sincsq(x-0.5/2)};
    \draw[ink, <->] (axis cs:-0.5/2,1.8) -- (axis cs:0.5/2,1.8);
    \node[ink, anchor=south, inner sep=2pt] at (axis cs:0,1.8) {$0.5\lambda/a$};
  \end{axis}
\end{tikzpicture}
\end{document}
