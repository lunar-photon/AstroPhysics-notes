\documentclass[tikz,border=6pt]{standalone}
\usepackage{amsmath,amssymb}
\usepackage{xcolor}
\definecolor{accent}{HTML}{A13D2C}
\definecolor{accent2}{HTML}{2B5F6B}
\definecolor{muted}{HTML}{5A564F}
\definecolor{panel}{HTML}{F8F6F0}
\definecolor{linecol}{HTML}{D8D2C4}
\definecolor{astroblue}{HTML}{1B4F72}
\definecolor{ink}{HTML}{1E1C1A}
\usetikzlibrary{calc,arrows.meta,angles,quotes}
\usepackage[T1]{fontenc}\usepackage{newpxtext,newpxmath}  % same fonts as the notes
\begin{document}
% A flat wave meets an opening between two baffles. Each open point sends out a wavelet; far away
% the wavelets leave along parallel rays at the angle theta. The tilted wavefront is perpendicular to
% the rays, and the thick segment is the extra path of the lowest wavelet, (top-to-bottom) times sin(theta).
\begin{tikzpicture}[font=\small, >={Stealth[length=2.2mm]}]
  \def\th{22}
  % incoming flat wave
  \foreach \x in {-3.2,-2.6,-2.0,-1.4,-0.8} \draw[linecol, line width=0.8pt] (\x,-1.8) -- (\x,1.8);
  \foreach \y in {-1.4,-0.7,0,0.7,1.4} \draw[astroblue, ->, line width=0.6pt] (-3.6,\y) -- (-0.2,\y);
  \node[muted, anchor=south] at (-2.0,1.85) {flat wave};
  % baffles
  \fill[muted!70] (-0.08,1.62) rectangle (0.08,2.5);
  \fill[muted!70] (-0.08,-1.62) rectangle (0.08,-2.5);
  \node[muted, anchor=east] at (-0.15,2.2) {baffle};
  \node[muted, anchor=east] at (-0.15,-2.2) {baffle};
  % wavelets
  \foreach \y in {-1.4,-0.7,0,0.7,1.4} {
    \foreach \R in {0.45,0.9} \draw[astroblue!45, line width=0.5pt] (0,\y) ++(-90:\R) arc[start angle=-90, end angle=90, radius=\R];
    \fill[ink] (0,\y) circle (0.05);
  }
  % rays at theta
  \foreach \y in {-1.4,-0.7,0,0.7,1.4} \draw[accent, ->, line width=0.8pt] (0,\y) -- ++(\th:4.6);
  \node[accent, anchor=west] at ($(0,1.4)+(\th:4.6)$) {ray at $\theta$};
  % tilted wavefront through the top point, perpendicular to the rays
  \draw[accent2, dashed, line width=0.8pt] (0,1.4) -- ++(\th-90:3.0) node[anchor=west, inner sep=3pt] {tilted wavefront};
  % extra path of the lowest wavelet: (2.8) sin(theta) along its ray
  \pgfmathsetmacro{\L}{2.8*sin(\th)}
  \draw[ink, line width=2.2pt] (0,-1.4) -- ++(\th:\L);
  \draw[ink] ($(0,-1.4)+(\th:\L/2)$) -- (1.0,-2.0) node[anchor=north west, inner sep=2pt] {extra path};
  % the angle theta at the middle point
  \draw[muted, dashed] (0,0) -- (1.6,0);
  \draw[muted] (1.25,0) arc[start angle=0, end angle=\th, radius=1.25];
  \node[muted] at ($(0,0)+(\th/2:1.5)$) {$\theta$};
  % opening label
\end{tikzpicture}
\end{document}
