% a mark at an observed line: dotted guide, tick above the panel, label on row 0 or row 1
\newcommand{\specmark}[3]{%
  \draw[linecol, dotted, line width=0.6pt] (axis cs:#1,-0.05) -- (axis cs:#1,1.1);
  \draw[muted, line width=0.6pt] (axis cs:#1,1.1) -- ([yshift=3pt+#2*11pt]axis cs:#1,1.1);
  \node[muted, font=\footnotesize, anchor=south, inner sep=1pt] at ([yshift=3pt+#2*11pt]axis cs:#1,1.1) {#3};}
% one panel: name, name of the data file, extra axis options
\newcommand{\specpanel}[3]{%
  \begin{axis}[width=12.5cm, height=2.3cm, scale only axis, xmin=380, xmax=920, ymin=-0.05, ymax=1.1,
      axis lines*=left, ytick=\empty, ylabel={#1}, xtick={400,500,600,700,800,900}, restrict y to domain*=-0.2:1.1,
      every axis label/.style={font=\small, text=muted}, tick style={muted}, clip=false, #3]
    \addplot[ink, line width=0.4pt] table[header=false, comment chars={\#}, x index=0, y index=1] {\datadir spec_#2.dat};
    \input{\datadir spec_ticks_#2.tex}
  \end{axis}}
