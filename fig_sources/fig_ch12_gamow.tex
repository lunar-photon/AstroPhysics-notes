\begin{tikzpicture}[font=\small]

  \begin{axis}[name=lin, width=12cm, height=5.2cm, scale only axis,

    \addplot[accent, line width=1.0pt, domain=0:20, samples=150] {exp(-x/\kT)};

    \node[muted, above] at (axis cs:\Ez,1.06) {$E_0=6.1\,\mathrm{keV}$};
